  <action>
Create a new canvas-based visualizer component that renders real-time audio data.

The component:
- Takes an AnalyserNode as a prop
- Draws either waveform (oscilloscope) or frequency bars based on a mode prop
- Uses requestAnimationFrame for smooth rendering
- Stops rendering when not playing (no idle draw loops)
- TE aesthetic: thin accent-colored lines on transparent background
- Responsive: sizes to container via CSS

```jsx
import React, { useRef, useEffect } from 'react';

/**
 * Visualizer — Canvas-based real-time audio waveform/frequency display.
 *
 * Connects to a Web Audio AnalyserNode and draws audio data on a canvas.
 * Designed to sit in the transport bar near Bonki.
 *
 * @param {Object} props
 * @param {AnalyserNode|null} props.analyser - Web Audio AnalyserNode
 * @param {boolean} props.isPlaying - Whether audio is currently playing
 * @param {'waveform'|'bars'} props.mode - Visualization mode
 * @param {string} props.color - Line/bar color (default: accent gold)
 */
function Visualizer({ analyser, isPlaying, mode = 'waveform', color = '#f5a623' }) {
  const canvasRef = useRef(null);
  const rafRef = useRef(null);

  useEffect(() => {
    if (!analyser || !isPlaying) {
      // Clear canvas when not playing
      const canvas = canvasRef.current;
      if (canvas) {
        const ctx = canvas.getContext('2d');
        ctx.clearRect(0, 0, canvas.width, canvas.height);
      }
      return;
    }

    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext('2d');
    const dpr = window.devicePixelRatio || 1;

    // Size canvas for retina
    const rect = canvas.getBoundingClientRect();
    canvas.width = rect.width * dpr;
    canvas.height = rect.height * dpr;
    ctx.scale(dpr, dpr);

    const bufferLength = analyser.frequencyBinCount;
    const dataArray = new Uint8Array(bufferLength);

    function drawWaveform() {
      analyser.getByteTimeDomainData(dataArray);
      const w = rect.width;
      const h = rect.height;

      ctx.clearRect(0, 0, w, h);

      // Draw center line (very subtle)
      ctx.strokeStyle = 'rgba(255, 255, 255, 0.05)';
      ctx.lineWidth = 1;
      ctx.beginPath();
      ctx.moveTo(0, h / 2);
      ctx.lineTo(w, h / 2);
      ctx.stroke();

      // Draw waveform
      ctx.strokeStyle = color;
      ctx.lineWidth = 1.5;
      ctx.beginPath();

      const sliceWidth = w / bufferLength;
      let x = 0;

      for (let i = 0; i < bufferLength; i++) {
        const v = dataArray[i] / 128.0;
        const y = (v * h) / 2;

        if (i === 0) {
          ctx.moveTo(x, y);
        } else {
          ctx.lineTo(x, y);
        }
        x += sliceWidth;
      }

      ctx.stroke();

      rafRef.current = requestAnimationFrame(drawWaveform);
    }

    function drawBars() {
      analyser.getByteFrequencyData(dataArray);
      const w = rect.width;
      const h = rect.height;
      const barCount = 24;
      const barWidth = w / barCount;
      const step = Math.floor(bufferLength / barCount);

      ctx.clearRect(0, 0, w, h);

      for (let i = 0; i < barCount; i++) {
        const value = dataArray[i * step];
        const barHeight = (value / 255) * h * 0.9;
        const x = i * barWidth;
        const y = h - barHeight;

        // Gradient from accent color to dim
        const alpha = 0.3 + (value / 255) * 0.7;
        ctx.fillStyle = color + Math.round(alpha * 255).toString(16).padStart(2, '0');
        ctx.fillRect(x + 1, y, barWidth - 2, barHeight);
      }

      rafRef.current = requestAnimationFrame(drawBars);
    }

    const draw = mode === 'bars' ? drawBars : drawWaveform;
    rafRef.current = requestAnimationFrame(draw);

    return () => {
      if (rafRef.current) cancelAnimationFrame(rafRef.current);
    };
  }, [analyser, isPlaying, mode, color]);

  return (
    <canvas
      ref={canvasRef}
      className="visualizer-canvas"
      aria-hidden="true"
    />
  );
}

export default Visualizer;
```

Add effect to create/connect analyser when playing starts:
```js
useEffect(() => {
  if (isPlaying && !analyser) {
    const audioCtx = getAudioContext();
    if (audioCtx) {
      const node = audioCtx.createAnalyser();
      node.fftSize = 512;
      node.smoothingTimeConstant = 0.8;
      // Connect to Strudel's destination to tap audio output
      // Strudel routes through audioCtx.destination, so we insert before it
      try {
        audioCtx.destination.connect && audioCtx.destination.connect(node);
      } catch {
        // If direct destination connect fails, try master gain
      }
      setAnalyser(node);
    }
  }
}, [isPlaying, analyser]);
```

NOTE: The AnalyserNode connection is tricky with Strudel. The simplest approach is:
- Get the AudioContext from Strudel
- Create an AnalyserNode
- Use audioCtx.destination as a pass-through (this may not work directly)

ALTERNATIVE APPROACH (more reliable): Create the analyser during initAudio and connect it as the last node before destination. This requires modifying main.jsx to insert the analyser into Strudel's audio chain.

If direct connection doesn't work, fall back to:
```js
// Create a gain node as a tap point
const tap = audioCtx.createGain();
tap.gain.value = 1;
tap.connect(audioCtx.destination);
tap.connect(node);
```
